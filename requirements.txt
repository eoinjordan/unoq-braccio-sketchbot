# numpy < 2 on cp313, so pinning `numpy<2` forces a slow source compile. We keep

# numpy-2 clean. On the device, `scripts/run_demo.sh setup` prefers Debian's
# python3-numpy / python3-opencv (prebuilt) so nothing has to compile at all.

# numpy-2 clean. On the UNO Q, Debian's python3-opencv (4.x) + opencv-data
# provide the same cascades.
opencv-python-headless>=4.8,<5
